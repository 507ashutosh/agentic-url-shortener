package com.agentic.shortener.ratelimit;

import java.io.IOException;
import java.time.Clock;

import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import com.agentic.shortener.config.ShortenerProperties;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

/** Limits how often one client can create links. Redirects and stats are not limited. */
@Component
public class RateLimitFilter extends OncePerRequestFilter {

    private final RateLimiter rateLimiter;

    public RateLimitFilter(ShortenerProperties properties, Clock clock) {
        this.rateLimiter = new RateLimiter(properties.rateLimit().requests(),
                properties.rateLimit().window(), clock);
    }

    @Override
    protected boolean shouldNotFilter(HttpServletRequest request) {
        return !("POST".equals(request.getMethod()) && "/shorten".equals(request.getRequestURI()));
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response,
                                    FilterChain chain) throws ServletException, IOException {
        // Remote address, not X-Forwarded-For: that header is client-controlled unless a trusted proxy sets it.
        if (rateLimiter.tryAcquire(request.getRemoteAddr())) {
            chain.doFilter(request, response);
            return;
        }
        response.setStatus(HttpStatus.TOO_MANY_REQUESTS.value());
        response.setHeader(HttpHeaders.RETRY_AFTER, String.valueOf(rateLimiter.windowSeconds()));
        response.setContentType(MediaType.APPLICATION_PROBLEM_JSON_VALUE);
        response.getWriter().write("""
                {"title":"Too Many Requests","status":429,"detail":"Rate limit exceeded. Try again later."}""");
    }
}

package com.agentic.shortener.ratelimit;

import java.time.Clock;
import java.time.Duration;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Fixed-window rate limiter: each key (client IP) gets {@code limit} requests per window.
 * State is in memory, so it applies per application instance.
 */
public class RateLimiter {

    private static final int CLEANUP_THRESHOLD = 10_000;

    private final int limit;
    private final long windowMillis;
    private final Clock clock;
    private final ConcurrentHashMap<String, Window> windows = new ConcurrentHashMap<>();

    private record Window(long startMillis, int count) {
    }

    public RateLimiter(int limit, Duration window, Clock clock) {
        this.limit = limit;
        this.windowMillis = window.toMillis();
        this.clock = clock;
    }

    /** Returns true if this request is within the limit for the key. */
    public boolean tryAcquire(String key) {
        long now = clock.millis();
        if (windows.size() > CLEANUP_THRESHOLD) {
            windows.values().removeIf(w -> isExpired(w, now));
        }
        // compute() is atomic per key, so concurrent requests from one client are counted correctly.
        Window window = windows.compute(key, (k, current) ->
                current == null || isExpired(current, now)
                        ? new Window(now, 1)
                        : new Window(current.startMillis(), current.count() + 1));
        return window.count() <= limit;
    }

    public long windowSeconds() {
        return Math.max(1, windowMillis / 1000);
    }

    private boolean isExpired(Window window, long now) {
        return now - window.startMillis() >= windowMillis;
    }
}

package com.agentic.shortener.service;

import java.security.SecureRandom;

import org.springframework.stereotype.Component;

/** Produces random 7-character codes from letters and digits (62^7, about 3.5 trillion values). */
@Component
public class CodeGenerator {

    static final int LENGTH = 7;
    private static final String ALPHABET = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz";

    // SecureRandom so codes cannot be guessed from earlier ones.
    private final SecureRandom random = new SecureRandom();

    public String next() {
        StringBuilder code = new StringBuilder(LENGTH);
        for (int i = 0; i < LENGTH; i++) {
            code.append(ALPHABET.charAt(random.nextInt(ALPHABET.length())));
        }
        return code.toString();
    }
}

package com.agentic.shortener.service;

import java.time.Clock;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.agentic.shortener.domain.UrlMapping;
import com.agentic.shortener.domain.UrlMappingRepository;

/** The business rules: create a short link, resolve it for a redirect, report its stats. */
@Service
public class UrlService {

    static final int MAX_CODE_ATTEMPTS = 5;

    private final UrlMappingRepository repository;
    private final UrlValidator validator;
    private final CodeGenerator codeGenerator;
    private final Clock clock;

    public UrlService(UrlMappingRepository repository, UrlValidator validator,
                      CodeGenerator codeGenerator, Clock clock) {
        this.repository = repository;
        this.validator = validator;
        this.codeGenerator = codeGenerator;
        this.clock = clock;
    }

    @Transactional
    public UrlMapping shorten(String url) {
        String validUrl = validator.validate(url);
        return repository.save(new UrlMapping(newUniqueCode(), validUrl, clock.instant()));
    }

    /** Returns the long URL for a code and counts the visit. */
    @Transactional
    public String resolve(String code) {
        UrlMapping mapping = find(code);
        repository.recordClick(code, clock.instant());
        return mapping.getOriginalUrl();
    }

    @Transactional(readOnly = true)
    public UrlMapping stats(String code) {
        return find(code);
    }

    private UrlMapping find(String code) {
        return repository.findByCode(code).orElseThrow(() -> new CodeNotFoundException(code));
    }

    // A random code can collide with an existing one; retry a bounded number of times.
    private String newUniqueCode() {
        for (int attempt = 0; attempt < MAX_CODE_ATTEMPTS; attempt++) {
            String code = codeGenerator.next();
            if (!repository.existsByCode(code)) {
                return code;
            }
        }
        throw new IllegalStateException("Could not generate a unique code after " + MAX_CODE_ATTEMPTS + " attempts");
    }
}

package com.agentic.shortener.service;

import java.net.URI;
import java.net.URISyntaxException;

import org.springframework.stereotype.Component;

/** Decides whether a submitted URL is acceptable to shorten. */
@Component
public class UrlValidator {

    static final int MAX_LENGTH = 2048;

    /** Returns the trimmed URL, or throws {@link InvalidUrlException} saying what is wrong. */
    public String validate(String url) {
        if (url == null || url.isBlank()) {
            throw new InvalidUrlException("url is required");
        }
        String trimmed = url.trim();
        if (trimmed.length() > MAX_LENGTH) {
            throw new InvalidUrlException("url must be at most " + MAX_LENGTH + " characters");
        }
        URI uri;
        try {
            uri = new URI(trimmed);
        } catch (URISyntaxException e) {
            throw new InvalidUrlException("url is not a valid URL");
        }
        // Only web links: blocks javascript:, file:, data: and similar schemes.
        String scheme = uri.getScheme();
        if (scheme == null || !(scheme.equalsIgnoreCase("http") || scheme.equalsIgnoreCase("https"))) {
            throw new InvalidUrlException("url must start with http:// or https://");
        }
        if (uri.getHost() == null || uri.getHost().isBlank()) {
            throw new InvalidUrlException("url must contain a host");
        }
        return trimmed;
    }
}

package com.agentic.shortener.web;

import java.net.URI;

import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

import com.agentic.shortener.config.ShortenerProperties;
import com.agentic.shortener.domain.UrlMapping;
import com.agentic.shortener.service.UrlService;
import com.agentic.shortener.web.dto.ShortenRequest;
import com.agentic.shortener.web.dto.ShortenResponse;
import com.agentic.shortener.web.dto.StatsResponse;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.Parameter;
import io.swagger.v3.oas.annotations.headers.Header;
import io.swagger.v3.oas.annotations.media.Content;
import io.swagger.v3.oas.annotations.media.Schema;
import io.swagger.v3.oas.annotations.responses.ApiResponse;
import io.swagger.v3.oas.annotations.tags.Tag;

@RestController
@Tag(name = "Short links")
public class UrlController {

    // Codes are letters and digits only, so /{code} never swallows other paths.
    private static final String CODE = "{code:[A-Za-z0-9]{4,10}}";

    private final UrlService urlService;
    private final ShortenerProperties properties;

    public UrlController(UrlService urlService, ShortenerProperties properties) {
        this.urlService = urlService;
        this.properties = properties;
    }

    @PostMapping("/shorten")
    @Operation(summary = "Create a short link")
    @ApiResponse(responseCode = "201", description = "Link created; Location holds the short URL")
    @ApiResponse(responseCode = "400", description = "Missing, malformed or non-http(s) URL",
            content = @Content(schema = @Schema(implementation = ProblemDetail.class)))
    @ApiResponse(responseCode = "429", description = "Too many links created from this IP; see Retry-After",
            headers = @Header(name = "Retry-After", description = "Seconds to wait"),
            content = @Content(schema = @Schema(implementation = ProblemDetail.class)))
    public ResponseEntity<ShortenResponse> shorten(@RequestBody ShortenRequest request) {
        UrlMapping mapping = urlService.shorten(request.url());
        String shortUrl = properties.baseUrl() + "/" + mapping.getCode();
        return ResponseEntity.created(URI.create(shortUrl))
                .body(new ShortenResponse(mapping.getCode(), shortUrl, mapping.getOriginalUrl()));
    }

    // 302 (temporary) rather than 301: browsers cache 301s, which would hide later clicks from the stats.
    @GetMapping("/" + CODE)
    @Operation(summary = "Follow a short link", description = """
            Redirects to the original URL and counts the click. This is the link visitors open.

            **"Try it out" here shows "Failed to fetch". That is expected, not an error:** the browser \
            follows the redirect to the original website, which does not allow Swagger UI's page to read \
            its response (CORS). The click is still counted.

            To test it, open `http://localhost:8080/{code}` in a browser tab, or run \
            `curl -i http://localhost:8080/{code}` to see the 302 and its `Location` header. \
            Use `GET /stats/{code}` to check the click count.""")
    @ApiResponse(responseCode = "302", description = "Redirect to the original URL",
            headers = @Header(name = "Location", description = "The original URL"))
    @ApiResponse(responseCode = "404", description = "Unknown code",
            content = @Content(schema = @Schema(implementation = ProblemDetail.class)))
    public ResponseEntity<Void> redirect(@Parameter(example = "Ab3dE9x") @PathVariable String code) {
        return ResponseEntity.status(HttpStatus.FOUND)
                .location(URI.create(urlService.resolve(code)))
                .build();
    }

    @GetMapping("/stats/" + CODE)
    @Operation(summary = "Click statistics for a short link")
    @ApiResponse(responseCode = "200", description = "Statistics")
    @ApiResponse(responseCode = "404", description = "Unknown code",
            content = @Content(schema = @Schema(implementation = ProblemDetail.class)))
    public StatsResponse stats(@Parameter(example = "Ab3dE9x") @PathVariable String code) {
        UrlMapping mapping = urlService.stats(code);
        return new StatsResponse(mapping.getCode(), mapping.getOriginalUrl(), mapping.getClickCount(),
                mapping.getCreatedAt(), mapping.getLastAccessedAt());
    }
}
