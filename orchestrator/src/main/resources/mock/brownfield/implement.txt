package com.agentic.shortener.domain;

import java.time.Instant;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.Table;

/** One short code and the long URL it points to. Maps to the url_mapping table. */
@Entity
@Table(name = "url_mapping")
public class UrlMapping {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, unique = true, length = 10)
    private String code;

    @Column(name = "original_url", nullable = false, length = 2048)
    private String originalUrl;

    @Column(name = "created_at", nullable = false)
    private Instant createdAt;

    @Column(name = "click_count", nullable = false)
    private long clickCount;

    @Column(name = "last_accessed_at")
    private Instant lastAccessedAt;

    // Null means the link never expires.
    @Column(name = "expires_at")
    private Instant expiresAt;

    protected UrlMapping() {
        // required by JPA
    }

    public UrlMapping(String code, String originalUrl, Instant createdAt) {
        this(code, originalUrl, createdAt, null);
    }

    public UrlMapping(String code, String originalUrl, Instant createdAt, Instant expiresAt) {
        this.code = code;
        this.originalUrl = originalUrl;
        this.createdAt = createdAt;
        this.expiresAt = expiresAt;
    }

    /** True once {@code now} has reached the expiry time. Links without an expiry never expire. */
    public boolean isExpiredAt(Instant now) {
        return expiresAt != null && !now.isBefore(expiresAt);
    }

    public Long getId() {
        return id;
    }

    public String getCode() {
        return code;
    }

    public String getOriginalUrl() {
        return originalUrl;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }

    public long getClickCount() {
        return clickCount;
    }

    public Instant getLastAccessedAt() {
        return lastAccessedAt;
    }

    public Instant getExpiresAt() {
        return expiresAt;
    }
}

package com.agentic.shortener.service;

import java.time.Clock;
import java.time.Duration;
import java.time.Instant;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.agentic.shortener.domain.UrlMapping;
import com.agentic.shortener.domain.UrlMappingRepository;

/** The business rules: create a short link, resolve it for a redirect, report its stats. */
@Service
public class UrlService {

    static final int MAX_CODE_ATTEMPTS = 5;
    static final int MAX_EXPIRY_DAYS = 3650;

    private final UrlMappingRepository repository;
    private final UrlValidator validator;
    private final CodeGenerator codeGenerator;
    private final Clock clock;

    public UrlService(UrlMappingRepository repository, UrlValidator validator,
                      CodeGenerator codeGenerator, Clock clock) {
        this.repository = repository;
        this.validator = validator;
        this.codeGenerator = codeGenerator;
        this.clock = clock;
    }

    /** Creates a link that never expires. */
    @Transactional
    public UrlMapping shorten(String url) {
        return shorten(url, null);
    }

    /** Creates a link; if {@code expiresInDays} is given, it stops working that many days from now. */
    @Transactional
    public UrlMapping shorten(String url, Integer expiresInDays) {
        String validUrl = validator.validate(url);
        Instant now = clock.instant();
        Instant expiresAt = expiryFrom(now, expiresInDays);
        return repository.save(new UrlMapping(newUniqueCode(), validUrl, now, expiresAt));
    }

    /** Returns the long URL for a code and counts the visit. Expired links are refused and not counted. */
    @Transactional
    public String resolve(String code) {
        UrlMapping mapping = find(code);
        Instant now = clock.instant();
        if (mapping.isExpiredAt(now)) {
            throw new LinkExpiredException(code);
        }
        repository.recordClick(code, now);
        return mapping.getOriginalUrl();
    }

    @Transactional(readOnly = true)
    public UrlMapping stats(String code) {
        return find(code);
    }

    private UrlMapping find(String code) {
        return repository.findByCode(code).orElseThrow(() -> new CodeNotFoundException(code));
    }

    private static Instant expiryFrom(Instant now, Integer expiresInDays) {
        if (expiresInDays == null) {
            return null;
        }
        if (expiresInDays < 1 || expiresInDays > MAX_EXPIRY_DAYS) {
            throw new InvalidExpiryException("expiresInDays must be between 1 and " + MAX_EXPIRY_DAYS);
        }
        return now.plus(Duration.ofDays(expiresInDays));
    }

    // A random code can collide with an existing one; retry a bounded number of times.
    private String newUniqueCode() {
        for (int attempt = 0; attempt < MAX_CODE_ATTEMPTS; attempt++) {
            String code = codeGenerator.next();
            if (!repository.existsByCode(code)) {
                return code;
            }
        }
        throw new IllegalStateException("Could not generate a unique code after " + MAX_CODE_ATTEMPTS + " attempts");
    }
}

package com.agentic.shortener.web;

import java.net.URI;
import java.time.Clock;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RestController;

import com.agentic.shortener.config.ShortenerProperties;
import com.agentic.shortener.domain.UrlMapping;
import com.agentic.shortener.service.UrlService;
import com.agentic.shortener.web.dto.ShortenRequest;
import com.agentic.shortener.web.dto.ShortenResponse;
import com.agentic.shortener.web.dto.StatsResponse;

@RestController
public class UrlController {

    // Codes are letters and digits only, so /{code} never swallows other paths.
    private static final String CODE = "{code:[A-Za-z0-9]{4,10}}";

    private final UrlService urlService;
    private final ShortenerProperties properties;
    private final Clock clock;

    public UrlController(UrlService urlService, ShortenerProperties properties, Clock clock) {
        this.urlService = urlService;
        this.properties = properties;
        this.clock = clock;
    }

    @PostMapping("/shorten")
    public ResponseEntity<ShortenResponse> shorten(@RequestBody ShortenRequest request) {
        UrlMapping mapping = urlService.shorten(request.url(), request.expiresInDays());
        String shortUrl = properties.baseUrl() + "/" + mapping.getCode();
        return ResponseEntity.created(URI.create(shortUrl))
                .body(new ShortenResponse(mapping.getCode(), shortUrl, mapping.getOriginalUrl(),
                        mapping.getExpiresAt()));
    }

    // 302 (temporary) rather than 301: browsers cache 301s, which would hide later clicks from the stats.
    @GetMapping("/" + CODE)
    public ResponseEntity<Void> redirect(@PathVariable String code) {
        return ResponseEntity.status(HttpStatus.FOUND)
                .location(URI.create(urlService.resolve(code)))
                .build();
    }

    @GetMapping("/stats/" + CODE)
    public StatsResponse stats(@PathVariable String code) {
        UrlMapping mapping = urlService.stats(code);
        return new StatsResponse(mapping.getCode(), mapping.getOriginalUrl(), mapping.getClickCount(),
                mapping.getCreatedAt(), mapping.getLastAccessedAt(),
                mapping.getExpiresAt(), mapping.isExpiredAt(clock.instant()));
    }
}

package com.agentic.shortener.web;

import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import com.agentic.shortener.service.CodeNotFoundException;
import com.agentic.shortener.service.InvalidExpiryException;
import com.agentic.shortener.service.InvalidUrlException;
import com.agentic.shortener.service.LinkExpiredException;

/** Turns exceptions into consistent JSON error bodies (RFC 9457 problem details). */
@RestControllerAdvice
public class ApiExceptionHandler {

    @ExceptionHandler(InvalidUrlException.class)
    ProblemDetail invalidUrl(InvalidUrlException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, e.getMessage());
    }

    @ExceptionHandler(InvalidExpiryException.class)
    ProblemDetail invalidExpiry(InvalidExpiryException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST, e.getMessage());
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    ProblemDetail unreadableBody(HttpMessageNotReadableException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.BAD_REQUEST,
                "Request body must be JSON like {\"url\": \"https://example.com\"}");
    }

    @ExceptionHandler(CodeNotFoundException.class)
    ProblemDetail notFound(CodeNotFoundException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.NOT_FOUND, e.getMessage());
    }

    @ExceptionHandler(LinkExpiredException.class)
    ProblemDetail expired(LinkExpiredException e) {
        return ProblemDetail.forStatusAndDetail(HttpStatus.GONE, e.getMessage());
    }
}
